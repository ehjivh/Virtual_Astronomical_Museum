        const daily = isDaily("toggle-s4");
        $("capSpec").textContent = daily ? "↑ 遠ざかる救急車。zが大きいほど右へ進み、後方の音波（波長）の間隔が広がります。" : "↑ 上：静止波長の基準線／下：観測された銀河スペクトル。重ねてズレ z を測定。";
        if (daily) drawSpectrumDaily();
        else drawSpectrumCosmic();

        const z = parseFloat(zval.value);
        const v = Math.round(C * z);
        $("zlab").textContent = z.toFixed(3);
        $("zout").textContent = z.toFixed(3);
        $("vout").textContent = v.toLocaleString() + " km/s";
      }

      function drawSpectrumCosmic() {
        const W = cvSpec.width,
          H = cvSpec.height;
        xs.clearRect(0, 0, W, H);
        const z = parseFloat(zval.value);

        const band = (y, shift, label) => {
          for (let px = 60; px < W - 20; px++) {
            const frac = (px - 60) / (W - 80),
              hue = 280 - frac * 280;
            xs.fillStyle = "hsl(" + hue + ",80%,55%)";
            xs.fillRect(px, y, 1, 26);
          }
          xs.fillStyle = "#000";
          [0.3, 0.55, 0.75].forEach((f) => {
            const fx = f + shift;
            if (fx < 1) xs.fillRect(60 + fx * (W - 80), y, 3, 26);
          });
          xs.fillStyle = COLORS.onSurface;
          xs.font = FONT;
          xs.fillText(label, 4, y + 16);
        };

        band(15, 0, "静止");
        band(75, z * 2.5, "観測");

        xs.strokeStyle = COLORS.tertiary;
        xs.lineWidth = 1.5;
        const x0 = 60 + 0.55 * (W - 80),
          x1p = 60 + (0.55 + z * 2.5) * (W - 80);
        xs.beginPath();
        xs.moveTo(x0, 58);
        xs.lineTo(x1p, 58);
        xs.stroke();
        xs.fillStyle = COLORS.tertiary;
        xs.fillText("→ 赤方偏移 z=" + z.toFixed(3), x0, 52);
      }

      // 日常ビュー（救急車のサイレン）
      function drawSpectrumDaily() {
        const W = cvSpec.width,
          H = cvSpec.height;
        xs.clearRect(0, 0, W, H);
        xs.fillStyle = COLORS.bgDeep;
        xs.fillRect(0, 0, W, H);
        const z = parseFloat(zval.value);

        // 道路
        xs.strokeStyle = COLORS.outlineV;
        xs.lineWidth = 2;
        xs.beginPath();
        xs.moveTo(10, H - 24);
        xs.lineTo(W - 10, H - 24);
        xs.stroke();

        // 観測者
        xs.fillStyle = COLORS.primary;
        xs.font = "18px serif";
        xs.fillText("🧑", 16, H - 30);
        xs.fillStyle = COLORS.outline;
        xs.font = "10px Roboto,sans-serif";
        xs.fillText("観測者", 14, H - 6);

        // 救急車の位置
        const carX = 70 + (z / 0.2) * (W - 200);
        const carY = H - 40;

        // 後方の音波
        const waveGap = 14 + (z / 0.2) * 40;
        for (let r = 0, k = 0; r < 300; r += waveGap, k++) {
          xs.strokeStyle = "rgba(255," + Math.round(120 - (z / 0.2) * 80) + "," + Math.round(120 - (z / 0.2) * 80) + "," + Math.max(0.15, 0.7 - k * 0.12) + ")";
          xs.lineWidth = 1.5;
          xs.beginPath();
          xs.arc(carX - 18, carY - 6, r, Math.PI * 0.6, Math.PI * 1.4);
          xs.stroke();
        }

        // 救急車ボディ
        xs.fillStyle = COLORS.onSurface;
        xs.fillRect(carX - 20, carY - 20, 40, 20);
        xs.fillStyle = "#ffb4ab";
        xs.fillRect(carX - 20, carY - 20, 40, 6);
        xs.fillStyle = COLORS.primary;
        xs.fillRect(carX + 4, carY - 16, 12, 10);
        xs.fillStyle = "#31353c";
        xs.beginPath();
        xs.arc(carX - 10, carY, 5, 0, TAU);
        xs.arc(carX + 10, carY, 5, 0, TAU);
        xs.fill();
        dot(xs, carX, carY - 24, 3, "#ff5252");

        xs.fillStyle = COLORS.tertiary;
        xs.font = FONT;
        xs.fillText("🚑 遠ざかる救急車　z=" + z.toFixed(3) + "（音＝光の波長が伸びる）", 70, 16);
      }

      on(zval, "input", drawSpectrum);

      // ---- ハッブル図 ----
      const cvH = $("cvHubble"),
        xh = cvH.getContext("2d");
      const hubblePts = [];

      function drawHubble() {
        const W = cvH.width,
          H = cvH.height;
        xh.clearRect(0, 0, W, H);
        drawAxis(xh, 50, 20, H - 30, W - 20);
        xh.fillStyle = COLORS.surfaceV;
        xh.font = FONT;
        xh.fillText("後退速度 v [km/s]", 56, 16);
        xh.fillText("距離 d [Mpc] →", W - 110, H - 10);

        const maxD = 900,
          maxV = 60000,
          px = (d) => 50 + (d / maxD) * (W - 70),
          py = (v) => H - 30 - (v / maxV) * (H - 50);

        let H0 = null;
        if (hubblePts.length >= 2) {
          // 最小二乗フィット（原点通過）
          let sxx = 0,
            sxy = 0;
          hubblePts.forEach((p) => {
            sxx += p.d * p.d;
            sxy += p.d * p.v;
          });
          H0 = sxy / sxx;
          xh.strokeStyle = COLORS.primary;
          xh.lineWidth = 2;
          xh.beginPath();
          xh.moveTo(px(0), py(0));
          xh.lineTo(px(maxD), py(H0 * maxD));
          xh.stroke();
        }

        hubblePts.forEach((p) => dot(xh, px(p.d), py(p.v), 5, COLORS.gold));

        $("ndots").textContent = hubblePts.length;
        const st = $("st4");
        if (H0 !== null && hubblePts.length >= 3) {
          $("H0out").textContent = H0.toFixed(1) + " km/s/Mpc";
          st.className = "status ok";
          st.textContent = "ハッブル定数 H₀≒" + H0.toFixed(1) + " km/s/Mpc を算出。距離梯子が完成しました！ この式で最果ての銀河までの距離も測定可能です。";
        } else {
          $("H0out").textContent = H0 ? H0.toFixed(1) : "–";
        }
      }

      on($("plotHubble"), "click", () => {
        const z = parseFloat(zval.value),
          v = C * z;
        const d = v / (H0_ASSUMED + (Math.random() * 8 - 4));
        hubblePts.push({ d: Math.min(d, 900), v: v });
        drawHubble();
      });

      function tick() {

          anim2 += 0.01;
          drawStep2();
        }

          if (snPlaying) {
            snPhase = Math.min(snPhase + 0.012, 1);
            if (snPhase >= 1) snPlaying = false;
          }
          drawStep3();
        }
        if ($("s4").classList.contains("active") && isDaily("toggle-s4")) {
          drawSpectrum();
        }
        requestAnimationFrame(tick);
      }
